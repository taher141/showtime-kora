// ==UserScript== // @name Block Popup Ads // @match *://*/* // @run-at document-start // @grant none // ==/UserScript== (function () { // منع فتح أي نافذة/تبويب جديد من الجافاسكربت window.open = function (url) { console.log('Blocked popup:', url); return null; }; // منع الروابط الخارجية اللي بتفتح في تبويب جديد document.addEventListener('click', function (e) { const a = e.target.closest('a'); if (a && a.target === '_blank' && a.hostname !== location.hostname) { e.preventDefault(); e.stopImmediatePropagation(); } }, true); })();
